${myFullRefLink} `; alert("مرحباً بك! تم تحديث نقاطك وبياناتك."); } catch (error) { console.error("Error:", error); alert("فشل الاتصال بالسيرفر. تأكد من تشغيل الـ Backend وصحة الرابط."); } }
